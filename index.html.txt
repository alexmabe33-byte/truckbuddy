    function initializeAcousticAudio() {
      if (audioCtx) return;
      
      try {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        
        // Setup analyser
        analyserNode = audioCtx.createAnalyser();
        analyserNode.fftSize = 256;
        
        // Setup main gain and misfire gain blocks
        mainGain = audioCtx.createGain();
        mainGain.gain.setValueAtTime(0.0, audioCtx.currentTime); // silent initially
        
        misfireGain = audioCtx.createGain();
        misfireGain.gain.setValueAtTime(1.0, audioCtx.currentTime);
        
        // Build Noise generator (essential for combustion rush & exhaust leaks)
        const bufferSize = audioCtx.sampleRate * 2;
        const noiseBuffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
        const output = noiseBuffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          output[i] = Math.random() * 2 - 1;
        }
        
        noiseNode = audioCtx.createBufferSource();
        noiseNode.buffer = noiseBuffer;
        noiseNode.loop = true;
        
        // Filter the noise to simulate combustion hiss or exhaust leaks
        noiseFilter = audioCtx.createBiquadFilter();
        noiseFilter.type = 'bandpass';
        noiseFilter.Q.setValueAtTime(1.5, audioCtx.currentTime);
        
        noiseGain = audioCtx.createGain();
        noiseGain.gain.setValueAtTime(0.04, audioCtx.currentTime); // low baseline hiss
        
        // Connect noise tree
        noiseNode.connect(noiseFilter);
        noiseFilter.connect(noiseGain);
        noiseGain.connect(misfireGain);
        
        // Assemble routing
        misfireGain.connect(mainGain);
        mainGain.connect(analyserNode);
        analyserNode.connect(audioCtx.destination);
        
        noiseNode.start(0);
        
        // Setup engine oscillators
        buildEngineOscillators();
        
        // Start live scope visuals loop
        requestAnimationFrame(renderVisualScopeFrame);
        
      } catch (err) {
        console.error("Failed to initialize audio contexts", err);
      }
    }

    function buildEngineOscillators() {
      if (!audioCtx) return;
      
      // Stop existing if any
      try {
        if (engineOsc1) engineOsc1.stop();
        if (engineOsc2) engineOsc2.stop();
        if (engineSubOsc) engineSubOsc.stop();
      } catch(e){}
      
      // Calculate fundamental firing frequency
      // For a 4-stroke engine: Freq = (RPM / 60) * (Cylinders / 2)
      let cylCount = 6;
      if (currentEngineType === 'v8') cylCount = 8;
      if (currentEngineType === 'flat4') cylCount = 4;
      
      let baseFreq = (currentRPM / 60) * (cylCount / 2);
      
      engineOsc1 = audioCtx.createOscillator();
      engineOsc1.type = 'sawtooth';
      engineOsc1.frequency.setValueAtTime(baseFreq, audioCtx.currentTime);
      
      engineOsc2 = audioCtx.createOscillator();
      engineOsc2.type = 'triangle';
      engineOsc2.frequency.setValueAtTime(baseFreq * 2.0, audioCtx.currentTime); // 2nd harmonic
      
      engineSubOsc = audioCtx.createOscillator();
      engineSubOsc.type = 'sawtooth';
      engineSubOsc.frequency.setValueAtTime(baseFreq * 0.5, audioCtx.currentTime); // Sub-harmonic combustion thud
      
      // Connect sound generators to misfire gain block
      engineOsc1.connect(misfireGain);
      engineOsc2.connect(misfireGain);
      engineSubOsc.connect(misfireGain);
      
      engineOsc1.start(0);
      engineOsc2.start(0);
      engineSubOsc.start(0);
      
      updateSynthFrequencies();
    }

    function updateSynthFrequencies() {
      if (!audioCtx || !engineOsc1) return;
      
      let cylCount = 6;
      if (currentEngineType === 'v8') cylCount = 8;
      if (currentEngineType === 'flat4') cylCount = 4;
      
      let baseFreq = (currentRPM / 60) * (cylCount / 2);
      let now = audioCtx.currentTime;
      
      engineOsc1.frequency.setTargetAtTime(baseFreq, now, 0.05);
      engineOsc2.frequency.setTargetAtTime(baseFreq * 1.5, now, 0.05); // Rich combustion harmonic
      engineSubOsc.frequency.setTargetAtTime(baseFreq * 0.5, now, 0.05);
      
      // Bandpass filter center frequency moves up with RPM
      let filterCenter = 150 + (currentRPM * 0.4);
      noiseFilter.frequency.setTargetAtTime(filterCenter, now, 0.05);
      
      // Fault adjustments
      if (currentFault === 'leak') {
        // High frequency exhaust blowing crack hiss
        noiseFilter.Q.setTargetAtTime(0.8, now, 0.1);
        noiseFilter.frequency.setTargetAtTime(2500, now, 0.1);
        noiseGain.gain.setTargetAtTime(0.35, now, 0.1);
      } else {
        noiseFilter.Q.setTargetAtTime(1.5, now, 0.1);
        noiseGain.gain.setTargetAtTime(0.06 + (currentLoad * 0.03), now, 0.1);
      }
    }

    function triggerRodKnock(time) {
      if (!audioCtx) return;
      
      // Spun rod bearing knock is a metallic "clack"
      // Trigger triangle pulse and highpass filtered noise pulse simultaneously
      let knockOsc = audioCtx.createOscillator();
      knockOsc.type = 'triangle';
      knockOsc.frequency.setValueAtTime(140, time);
      
      let knockFilter = audioCtx.createBiquadFilter();
      knockFilter.type = 'bandpass';
      knockFilter.frequency.setValueAtTime(900, time);
      knockFilter.Q.setValueAtTime(2.0, time);
      
      let knockGain = audioCtx.createGain();
      // Louder knock under high loading
      let knockVol = 0.18 + (currentLoad * 0.08);
      knockGain.gain.setValueAtTime(knockVol, time);
      knockGain.gain.exponentialRampToValueAtTime(0.001, time + 0.025); // very fast decay
      
      knockOsc.connect(knockFilter);
      knockFilter.connect(knockGain);
      knockGain.connect(analyserNode);
      
      knockOsc.start(time);
      knockOsc.stop(time + 0.03);
    }

    function renderVisualScopeFrame() {
      requestAnimationFrame(renderVisualScopeFrame);
      
      if (!scopeCtx || !analyserNode) return;
      
      const width = scopeCanvas.width;
      const height = scopeCanvas.height;
      
      // Soft phosphor clear trail (creates classic CRT glow trails)
      scopeCtx.fillStyle = 'rgba(0, 0, 0, 0.25)';
      scopeCtx.fillRect(0, 0, width, height);
      
      // Technical coordinates grid
      scopeCtx.strokeStyle = 'rgba(0, 255, 102, 0.05)';
      scopeCtx.lineWidth = 1;
      for (let x = 0; x < width; x += 30) {
        scopeCtx.beginPath();
        scopeCtx.moveTo(x, 0);
        scopeCtx.lineTo(x, height);
        scopeCtx.stroke();
      }
      for (let y = 0; y < height; y += 30) {
        scopeCtx.beginPath();
        scopeCtx.moveTo(0, y);
        scopeCtx.lineTo(width, y);
        scopeCtx.stroke();
      }
      
      // Gather audio buffer
      const bufferLength = analyserNode.frequencyBinCount;
      const dataArray = new Uint8Array(bufferLength);
      analyserNode.getByteTimeDomainData(dataArray);
      
      // Draw neon wave line
      scopeCtx.strokeStyle = getComputedStyle(document.body).getPropertyValue('--accent-color').trim() || '#00ff66';
      scopeCtx.lineWidth = 2.5;
      scopeCtx.shadowBlur = 6;
      scopeCtx.shadowColor = scopeCtx.strokeStyle;
      
      scopeCtx.beginPath();
      const sliceWidth = width / bufferLength;
      let x = 0;
      
      for (let i = 0; i < bufferLength; i++) {
        const v = dataArray[i] / 128.0;
        const y = (v * height) / 2;
        
        if (i === 0) {
          scopeCtx.moveTo(x, y);
        } else {
          scopeCtx.lineTo(x, y);
        }
        
        x += sliceWidth;
      }
      
      scopeCtx.lineTo(width, height / 2);
      scopeCtx.stroke();
      scopeCtx.shadowBlur = 0; // reset
      
      // Real-time audio sequencing loop inside requestAnimationFrame (crucial for exact timing!)
      if (isSynthRunning && audioCtx) {
        let now = audioCtx.currentTime;
        let revSec = currentRPM / 60;
        
        // Rod knock loop - 1 hit per revolution
        let knockInterval = 1 / revSec;
        if (currentFault === 'knock' && now - lastKnockTime >= knockInterval) {
          triggerRodKnock(now);
          lastKnockTime = now;
        }
        
        // Misfire loop - drops engine volume every 2nd engine revolution
        let misfireInterval = 2 / revSec;
        if (currentFault === 'misfire' && now - lastMisfireTime >= misfireInterval) {
          let dropDuration = 0.45 / revSec; // drop half a revolution
          misfireGain.gain.setValueAtTime(1.0, now);
          misfireGain.gain.exponentialRampToValueAtTime(0.08, now + 0.015);
          misfireGain.gain.setValueAtTime(0.08, now + dropDuration - 0.015);
          misfireGain.gain.exponentialRampToValueAtTime(1.0, now + dropDuration);
          lastMisfireTime = now;
        }
      }
      
      // Spin the mechanical piston assembly in About tab at proportional speed
      animateAboutPistonAssembly();
      
      // Update Radar dial on sidebar
      renderRotaryRadarFrame();
    }

    function renderRotaryRadarFrame() {
      if (!radarCtx) return;
      const size = 160;
      const center = size / 2;
      const radius = size / 2 - 5;
      
      radarCtx.fillStyle = 'rgba(0, 0, 0, 0.15)'; // trails
      radarCtx.fillRect(0, 0, size, size);
      
      // Outer radar dial shell
      let activeColor = getComputedStyle(document.body).getPropertyValue('--border-color').trim() || '#00ff66';
      radarCtx.strokeStyle = activeColor;
      radarCtx.lineWidth = 1;
      radarCtx.beginPath();
      radarCtx.arc(center, center, radius, 0, 2 * Math.PI);
      radarCtx.stroke();
      
      // Inner concentric coordinate circles
      radarCtx.strokeStyle = 'rgba(0, 255, 102, 0.08)';
      radarCtx.beginPath();
      radarCtx.arc(center, center, radius * 0.6, 0, 2 * Math.PI);
      radarCtx.arc(center, center, radius * 0.3, 0, 2 * Math.PI);
      radarCtx.stroke();
      
      // Crosshairs
      radarCtx.beginPath();
      radarCtx.moveTo(5, center); radarCtx.lineTo(size - 5, center);
      radarCtx.moveTo(center, 5); radarCtx.lineTo(center, size - 5);
      radarCtx.stroke();
      
      // Rotating scan sweeps
      radarAngle += 0.04;
      if (radarAngle >= 2 * Math.PI) radarAngle = 0;
      
      radarCtx.strokeStyle = activeColor;
      radarCtx.lineWidth = 2;
      radarCtx.beginPath();
      radarCtx.moveTo(center, center);
      let scanX = center + radius * Math.cos(radarAngle);
      let scanY = center + radius * Math.sin(radarAngle);
      radarCtx.lineTo(scanX, scanY);
      radarCtx.stroke();
      
      // Dynamic simulated signal blips (flickers based on engine sound)
      if (isSynthRunning) {
        radarCtx.fillStyle = '#ff3333';
        if (currentFault === 'healthy') radarCtx.fillStyle = activeColor;
        
        let offsetPulse = 1.0;
        if (analyserNode) {
          const arr = new Uint8Array(16);
          analyserNode.getByteFrequencyData(arr);
          offsetPulse = 1.0 + (arr[2] / 255.0) * 0.3;
        }
        
        radarCtx.beginPath();
        radarCtx.arc(center + 30 * Math.sin(radarAngle * 0.5), center - 20 * Math.cos(radarAngle * 0.8), 4 * offsetPulse, 0, 2 * Math.PI);
        radarCtx.fill();
      }
    }

    let crankAngle = 0;
    function animateAboutPistonAssembly() {
      const pGroup = document.getElementById('movingGroup');
      const rod = document.getElementById('connectingRod');
      const crank = document.getElementById('crankRotator');
      const spark = document.getElementById('pistonSpark');
      
      if (!pGroup || !rod || !crank) return;
      
      // Speed multiplier proportional to RPM
      let radsPerSec = (currentRPM / 60) * (2 * Math.PI);
      // Normalized frame rate speed offset
      crankAngle += (radsPerSec / 60); 
      if (crankAngle >= 4 * Math.PI) crankAngle = 0; // 720 degrees for full 4-stroke cycle
      
      // Crank radius and Rod length coordinate equations
      let r = 24;
      let L = 75;
      
      // Rotator angle
      crank.setAttribute('transform', `translate(100, 180) rotate(${-crankAngle * 180 / Math.PI})`);
      
      // Pin coordinate offsets
      let cx = r * Math.cos(crankAngle);
      let cy = r * Math.sin(crankAngle);
      
      // Pin strictly vertical constraint solver
      let py = cy + Math.sqrt(L*L - cx*cx);
      
      // Offset translation limits (0 coordinate is centered at x=100, y=180)
      let pistonOffsetTranslate = -py + 90;
      pGroup.setAttribute('transform', `translate(0, ${pistonOffsetTranslate})`);
      
      // Move connecting rod lines
      rod.setAttribute('x1', 100);
      rod.setAttribute('y1', 70 + pistonOffsetTranslate);
      rod.setAttribute('x2', 100 + cx);
      rod.setAttribute('y2', 180 - cy); // invert y for SVG coordinate orientation
      
      // Spark trigger logic: fires once every 2 crankshaft revolutions at TDC (Top Dead Center)
      // Represented by crankAngle near 2*PI (ignition stroke)
      let tolerance = 0.12;
      let ignitionThreshold = 2 * Math.PI;
      if (Math.abs(crankAngle - ignitionThreshold) < tolerance) {
        spark.style.display = 'block';
      } else {
        spark.style.display = 'none';
      }
    }

    function playUiBeep(freq, duration, type = 'sine') {
      if (!audioCtx) return;
      try {
        let osc = audioCtx.createOscillator();
        let gain = audioCtx.createGain();
        osc.type = type;
        osc.frequency.setValueAtTime(freq, audioCtx.currentTime);
        gain.gain.setValueAtTime(0.06, audioCtx.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);
        osc.connect(gain);
        gain.connect(audioCtx.destination);
        osc.start();
        osc.stop(audioCtx.currentTime + duration);
      } catch(e){}
    }

    function playFloppyDriveNoise() {
      if (!audioCtx) return;
      try {
        // Synthesize classic grinding mechanical stepper motor sound in Web Audio!
        let now = audioCtx.currentTime;
        
        for (let i = 0; i < 6; i++) {
          let clickTime = now + (i * 0.15);
          
          let noiseOsc = audioCtx.createOscillator();
          noiseOsc.type = 'sawtooth';
          noiseOsc.frequency.setValueAtTime(80, clickTime);
          
          let noiseFilt = audioCtx.createBiquadFilter();
          noiseFilt.type = 'bandpass';
          noiseFilt.frequency.setValueAtTime(350, clickTime);
          
          let noiseG = audioCtx.createGain();
          noiseG.gain.setValueAtTime(0.12, clickTime);
          noiseG.gain.exponentialRampToValueAtTime(0.001, clickTime + 0.08);
          
          noiseOsc.connect(noiseFilt);
          noiseFilt.connect(noiseG);
          noiseG.connect(audioCtx.destination);
          
          noiseOsc.start(clickTime);
          noiseOsc.stop(clickTime + 0.1);
        }
      } catch(e){}
    }
