    // Calculate Dynamic Distance from Pickup & Drop-off addresses
    function calculateAddressDistance() {
      const pickup = document.getElementById('pickup-select').value;
      const dropoff = document.getElementById('dropoff-select').value;

      if (pickup === dropoff) {
        currentDistanceMiles = 3.5; // Same zone short haul default
      } else {
        const key1 = `${pickup}-${dropoff}`;
        const key2 = `${dropoff}-${pickup}`;
        currentDistanceMiles = distanceMatrix[key1] || distanceMatrix[key2] || 18.5;
      }

      // Calculate estimated time (avg speed ~ 45 mph)
      const minutes = Math.round((currentDistanceMiles / 45) * 60);

      // Update UI displays
      document.getElementById('calculated-miles').innerText = `${currentDistanceMiles.toFixed(1)} Miles`;
      document.getElementById('calculated-time').innerText = `${minutes} Mins`;
      document.getElementById('map-tag-distance').innerText = `${currentDistanceMiles.toFixed(1)} miles calculated`;

      recalculatePrice();
    }

    // Preset Route Buttons
    function presetRoute(type) {
      if (type === 'short') {
        document.getElementById('pickup-select').value = 'houston_dt';
        document.getElementById('dropoff-select').value = 'pasadena_tx';
      } else if (type === 'medium') {
        document.getElementById('pickup-select').value = 'houston_dt';
        document.getElementById('dropoff-select').value = 'katy_tx';
      } else if (type === 'long') {
        document.getElementById('pickup-select').value = 'katy_tx';
        document.getElementById('dropoff-select').value = 'woodlands_tx';
      }
      calculateAddressDistance();
    }

    // Select Truck Tier
    function selectTruckTier(tier, basePrice, perMile) {
      selectedTruckBase = basePrice;
      selectedTruckPerMile = perMile;

      // Update visual styles on tier cards
      document.querySelectorAll('.truck-tier-card').forEach(card => {
        card.className = "truck-tier-card cursor-pointer p-4 rounded-xl border border-slate-800 bg-slate-950/50 hover:border-brand-500/50 transition relative group";
      });

      const selectedCard = document.getElementById(`tier-${tier}`);
      if (selectedCard) {
        selectedCard.className = "truck-tier-card cursor-pointer p-4 rounded-xl border-2 border-brand-500 bg-brand-500/10 transition relative group shadow-lg shadow-brand-500/5";
      }

      if (tier === 'compact') selectedTruckName = "Compact Truck";
      if (tier === 'fullsize') selectedTruckName = "Full-size Truck";
      if (tier === 'heavyduty') selectedTruckName = "Heavy Duty Truck";

      recalculatePrice();
    }

    function updateTrailerSelection(name, baseFee, mileFee) {
      selectedTrailerName = name;
      selectedTrailerBase = baseFee;
      selectedTrailerPerMile = mileFee;

      recalculatePrice();

    // Toggle Checkbox Addons
    function toggleAddon(type, price) {
      const checkboxes = document.querySelectorAll('.addon-checkbox');
      let sum = 0;
      checkboxes.forEach(cb => {
        if (cb.checked) {
          sum += 10; // Simple simulation total based on selections
        }
      });
      selectedAddonsTotal = sum;
      recalculatePrice();
    }

    // Master Price Recalculation Engine
    function recalculatePrice() {
      const truckDistanceCost = currentDistanceMiles * selectedTruckPerMile;
      const truckTotalCost = selectedTruckBase + truckDistanceCost;

      const trailerDistanceCost = currentDistanceMiles * selectedTrailerPerMile;
      const trailerTotalCost = selectedTrailerBase + trailerDistanceCost;

      const serviceFee = 6.50;
      const grandTotal = truckTotalCost + trailerTotalCost + selectedAddonsTotal + serviceFee;

      // Update Summary UI
      document.getElementById('summary-truck-label').innerText = `${selectedTruckName} Base Fee`;
      document.getElementById('summary-truck-cost').innerText = `$${selectedTruckBase.toFixed(2)}`;

      document.getElementById('summary-mileage-label').innerText = `Distance Fee (${currentDistanceMiles.toFixed(1)} mi @ $${selectedTruckPerMile.toFixed(2)}/mi)`;
      document.getElementById('summary-mileage-cost').innerText = `$${truckDistanceCost.toFixed(2)}`;

      const trailerRow = document.getElementById('row-trailer');
      if (selectedTrailerBase > 0) {
        trailerRow.style.display = 'flex';
        document.getElementById('summary-trailer-label').innerText = `${selectedTrailerName}`;
        document.getElementById('summary-trailer-cost').innerText = `$${trailerTotalCost.toFixed(2)}`;
      } else {
        trailerRow.style.display = 'none';
      }

      document.getElementById('summary-addons-cost').innerText = `$${selectedAddonsTotal.toFixed(2)}`;
      document.getElementById('summary-total-price').innerText = `$${grandTotal.toFixed(2)}`;
    }

      document.getElementById('view-driver').classList.add('hidden');

        document.getElementById('nav-customer').className = "px-3 py-1.5 rounded-lg flex items-center space-x-2 transition-all bg-brand-500 text-white font-semibold shadow-md";
      } else if (role === 'driver') {
        document.getElementById('view-driver').classList.remove('hidden');
        document.getElementById('nav-driver').className = "px-3 py-1.5 rounded-lg flex items-center space-x-2 transition-all bg-brand-500 text-white font-semibold shadow-md";

        document.getElementById('view-admin').classList.remove('hidden');
        document.getElementById('nav-admin').className = "px-3 py-1.5 rounded-lg flex items-center space-x-2 transition-all bg-brand-500 text-white font-semibold shadow-md";
      }
    }

    // Start Simulation Dispatch
    function startDispatchSimulation() {
      document.getElementById('view-customer').classList.add('hidden');
      document.getElementById('view-dispatch').classList.remove('hidden');
      simulationStep = 1;
      updateSimulationUI();
    }

    // Advance Step in Delivery Simulation
    function advanceSimulationStep() {
      simulationStep++;
      if (simulationStep > 4) simulationStep = 1;
      updateSimulationUI();
    }

    function updateSimulationUI() {
      const fill = document.getElementById('progress-fill');
      const badge = document.getElementById('dispatch-status-badge');
      const title = document.getElementById('dispatch-title');
      const truckNode = document.getElementById('sim-truck-node');
      const photoBox = document.getElementById('photo-proof-container');
      const photoTag = document.getElementById('photo-status-tag');

      if (simulationStep === 1) {
        fill.style.width = '25%';
        badge.innerText = 'Driver Dispatched';
        title.innerText = 'Driver Assigned & En Route to Pickup';
        truckNode.setAttribute('transform', 'translate(60, 220)');
      } else if (simulationStep === 2) {
        fill.style.width = '50%';
        badge.innerText = 'At Pickup Location';
        title.innerText = 'Driver Arrived - Loading Cargo';
        truckNode.setAttribute('transform', 'translate(180, 150)');

        photoTag.innerText = 'Pickup Photo Verified';
        photoTag.className = 'text-[10px] text-emerald-400 font-bold';
        photoBox.innerHTML = `
          <img src="https://images.unsplash.com/photo-1586528116311-ad8dd3c8310d?w=400&auto=format&fit=crop&q=80" class="w-full h-full object-cover rounded-lg" alt="Cargo Photo">
        `;
      } else if (simulationStep === 3) {
        fill.style.width = '75%';
        badge.innerText = 'In Transit';
        title.innerText = 'Truck & Trailer In Transit to Destination';
        truckNode.setAttribute('transform', 'translate(320, 100)');
      } else if (simulationStep === 4) {
        fill.style.width = '100%';
        badge.innerText = 'Haul Completed';
        title.innerText = 'Cargo Delivered & Completed!';
        truckNode.setAttribute('transform', 'translate(440, 200)');
      }

      // Highlight Icons
      for (let i = 1; i <= 4; i++) {
        const icon = document.getElementById(`step-${i}-icon`);
        if (i <= simulationStep) {
          icon.className = "w-8 h-8 rounded-full bg-brand-500 text-white flex items-center justify-center text-xs font-bold ring-4 ring-slate-900";
        } else {
          icon.className = "w-8 h-8 rounded-full bg-slate-800 text-slate-400 flex items-center justify-center text-xs font-bold ring-4 ring-slate-900";
        }
      }
    }
